cmake_minimum_required(VERSION 3.16)
project(ytplaylist VERSION 1.0.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(ytplaylist src/main.cpp)
target_compile_features(ytplaylist PRIVATE cxx_std_17)
target_link_libraries(ytplaylist PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(ytplaylist PRIVATE /W4 /utf-8)
  set_property(TARGET ytplaylist PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
  target_compile_options(ytplaylist PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(ytplaylist PRIVATE -municode)
else()
  target_compile_options(ytplaylist PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(BUILD_WEB "Build the C++ website server" ON)
if(BUILD_WEB)
  set(HEADERS_DIR "${CMAKE_BINARY_DIR}/headers")
  file(MAKE_DIRECTORY "${HEADERS_DIR}")
  function(download_header name url sha)
    if(NOT EXISTS "${HEADERS_DIR}/${name}")
      file(DOWNLOAD "${url}" "${HEADERS_DIR}/${name}" EXPECTED_HASH "SHA256=${sha}" TLS_VERIFY ON)
    endif()
    file(SHA256 "${HEADERS_DIR}/${name}" actual)
    if(NOT actual STREQUAL sha)
      message(FATAL_ERROR "Header checksum mismatch: ${name}")
    endif()
  endfunction()
  download_header(httplib.h "https://raw.githubusercontent.com/yhirose/cpp-httplib/v0.59.0/httplib.h" "dc1e4de3e0ef3a18f3a5818fab563eb3fa45d5d3bb1db527a8d389077ed5ee6c")
  download_header(json.hpp "https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp" "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  add_executable(ytplaylist-web src/server.cpp)
  target_compile_features(ytplaylist-web PRIVATE cxx_std_17)
  target_include_directories(ytplaylist-web PRIVATE "${HEADERS_DIR}")
  target_link_libraries(ytplaylist-web PRIVATE Threads::Threads)
  if(WIN32)
    target_link_libraries(ytplaylist-web PRIVATE ws2_32)
  endif()
  if(MSVC)
    target_compile_options(ytplaylist-web PRIVATE /W4 /utf-8)
    set_property(TARGET ytplaylist-web PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  install(TARGETS ytplaylist-web RUNTIME DESTINATION bin)
  install(DIRECTORY web DESTINATION bin)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(fake-downloader tests/fake_downloader.cpp)
  target_compile_features(fake-downloader PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(fake-downloader PRIVATE -municode)
  endif()
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
    $<TARGET_FILE:ytplaylist> $<TARGET_FILE:fake-downloader>)
  if(BUILD_WEB)
    add_executable(zip-fixture tests/zip_fixture.cpp)
    target_compile_features(zip-fixture PRIVATE cxx_std_17)
    add_test(NAME zip64 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_zip.py $<TARGET_FILE:zip-fixture>)
    add_test(NAME web COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_server.py
      $<TARGET_FILE:ytplaylist-web> $<TARGET_FILE:fake-downloader> ${CMAKE_CURRENT_SOURCE_DIR}/web)
  endif()
endif()
install(TARGETS ytplaylist RUNTIME DESTINATION bin)
