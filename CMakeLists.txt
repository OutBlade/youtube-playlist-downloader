cmake_minimum_required(VERSION 3.16)
project(ytplaylist VERSION 1.0.0 LANGUAGES CXX)
add_executable(ytplaylist src/main.cpp)
target_compile_features(ytplaylist PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ytplaylist PRIVATE /W4 /utf-8)
elseif(MINGW)
  target_compile_options(ytplaylist PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(ytplaylist PRIVATE -municode)
else()
  target_compile_options(ytplaylist PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(fake-downloader tests/fake_downloader.cpp)
  target_compile_features(fake-downloader PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(fake-downloader PRIVATE -municode)
  endif()
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
    $<TARGET_FILE:ytplaylist> $<TARGET_FILE:fake-downloader>)
endif()
install(TARGETS ytplaylist RUNTIME DESTINATION bin)
